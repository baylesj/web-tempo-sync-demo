    01two tabs

    In stock browsers where navigator.tempoSync is not present, the page uses a BroadcastChannel stand-in with the same API surface. No custom browser build is needed.

    1. Click Connect in the sequencer view (the api row shows stand-in (BroadcastChannel)).
    2. Click Open 2nd tab (or open this URL in another tab and click Connect).
    3. Pick different patterns in each tab (for example, four on the floor in tab 1 and syncopated in tab 2) and press Play or the space bar.
    4. Start and stop wait for the next quantum downbeat, and moving the tempo slider updates both tabs at once.

    02ableton live

    In the prototype Chromium build (macOS arm64 download, release v0.1.0), the browser process joins the local Ableton Link UDP multicast group (224.76.78.75:20808) and syncs with Ableton Live on the same laptop or local network. 

    xattr -cr Chromium.app
./Chromium.app/Contents/MacOS/Chromium \
  --enable-blink-features=TempoSync \
  --tempo-sync-engine=rust \
  "https://bayl.es/tempo-sync-demo/?autoconnect=1"
    1. In Ableton Live 12, open Settings > Link, Tempo & MIDI and enable Show Link Toggle and Start/Stop Sync.
    2. Click the Link button in the top-left corner of Live.
    3. In Chromium, check that api reads navigator.tempoSync and peers increments.

    03ableton move

    You do not need a Wi-Fi router to sync with an Ableton Move on the road. In Standalone Mode, Move exposes a USB-C network interface (http://move.local) that carries Ableton Link traffic over a single USB-C cable.

    1. On Ableton Move, press Shift + Step 2 (Setup), set Mode: Standalone (not Control Live), and turn Link: On and Sync Start/Stop: On.
    2. Plug Move's USB-C port (next to the power port, not the USB-A port) directly into your MacBook Pro.
    3. Launch the prototype Chromium build after plugging Move in so the Rust engine binds to the USB network interface alongside en0 and lo0.
    4. Chromium, Ableton Live, and Ableton Move discover each other and lock tempo, beat phase, and start/stop over USB.

    04wi-fi & latency

    To sync wirelessly across multiple laptops, iPads, or hardware units, or to calibrate browser output latency:

    • Shared Wi-Fi or phone hotspot: Connect devices to a local network or phone hotspot that allows peer-to-peer UDP multicast (client isolation off).
    • Turn off corporate VPNs: VPN tunnels often intercept or drop multicast packets on 224.76.78.75:20808.
    • Measure audio output offset: Enable measure output latency in section 03 latency of the sequencer view and record this browser's output alongside Live's metronome in a DAW.

    01audio clock

    Passing an AudioContext to session.captureState(audioContext) projects the shared session timeline directly into AudioContext.currentTime seconds, including output latency compensation: 

    const audioContext = new AudioContext();
const session = await navigator.tempoSync.requestSession({
  quantum: 4,
  startStopSync: true,
});

let nextStep = null; // the next 16th-note step, in beats, not yet scheduled

// Runs every 25 ms. Each call schedules only the steps it hasn't scheduled before, so a step
// that falls inside several overlapping lookahead windows still plays once.
function scheduleLookahead() {
  const state = session.captureState(audioContext);
  const now = audioContext.currentTime;
  const horizon = now + 0.100; // 100 ms lookahead window
  const quantum = session.quantum;

  if (nextStep === null) {
    nextStep = Math.ceil(state.beatAtTime(now, quantum) * 4) / 4;
  }
  const endBeat = state.beatAtTime(horizon, quantum);

  for (; nextStep < endBeat; nextStep += 0.25) {
    triggerStepAtTime(nextStep, state.timeAtBeat(nextStep, quantum));
  }
}

setInterval(scheduleLookahead, 25);

    02transport & ui

    Calling session.captureState() with no arguments uses performance.now() / 1000 for requestAnimationFrame visuals and transport countdowns: 

    // Quantize Play/Stop to the next quantum boundary
function toggleQuantizedPlay() {
  const state = session.captureState(audioContext);
  const now = audioContext.currentTime;
  const quantum = session.quantum;
  const nextBar = (Math.floor(state.beatAtTime(now, quantum) / quantum) + 1) * quantum;
  const at = state.timeAtBeat(nextBar, quantum);

  state.setIsPlaying(!state.isPlaying, at);
  session.commitState(state);
}

// 60 fps phase ring
function renderLoop() {
  const state = session.captureState();
  const phase = state.phaseAtTime(performance.now() / 1000, session.quantum);
  drawBeatRing(phase, state.tempo, state.isPlaying);
  requestAnimationFrame(renderLoop);
}

    03webidl

    The browser owns UDP multicast sockets, peer discovery, and clock-offset filtering in the browser process. Pages only observe beat/phase math and peer counts: 

    [SecureContext, Exposed=Window]
partial interface Navigator {
  [SameObject] readonly attribute TempoSync tempoSync;
};

[SecureContext, Exposed=Window]
interface TempoSync {
  Promise<TempoSession> requestSession(optional TempoSessionOptions options = {});
};

dictionary TempoSessionOptions {
  double quantum = 4.0;
  boolean startStopSync = false;
};

[SecureContext, Exposed=Window]
interface TempoSession : EventTarget {
  readonly attribute unsigned long numPeers;
  readonly attribute double quantum;
  readonly attribute boolean isStartStopSyncEnabled;

  TempoSessionState captureState(optional BaseAudioContext? audioContext = null);
  undefined commitState(TempoSessionState state);
  undefined close();

  attribute EventHandler onnumpeerschange;
  attribute EventHandler ontempochange;
  attribute EventHandler onplayingchange;
};

[SecureContext, Exposed=Window]
interface TempoSessionState {
  readonly attribute double tempo;
  readonly attribute boolean isPlaying;

  double beatAtTime(double time, double quantum);
  double phaseAtTime(double time, double quantum);
  double timeAtBeat(double beat, double quantum);
  readonly attribute double timeForIsPlaying;

  undefined setTempo(double bpm, double atTime);
  undefined requestBeatAtTime(double beat, double time, double quantum);
  undefined forceBeatAtTime(double beat, double time, double quantum);
  undefined setIsPlaying(boolean isPlaying, double atTime);
  undefined requestBeatAtStartPlayingTime(double beat, double quantum);
};
